import React, { useState } from 'react';

### 2. صفحه باغ عاطفی (Garden.js)
```javascript
import React, { useState, useEffect } from 'react';
import { useAuth } from '../context/AuthContext';
import InteractionService from '../services/interaction.service';
import GardenVisualization from '../components/GardenVisualization';

function Garden() {
  const { currentUser } = useAuth();
  const [gardenState, setGardenState] = useState({
    positiveCount: 0,
    negativeCount: 0,
    ratio: 0,
    healthLevel: 0 // 0-100
  });

  useEffect(() => {
    const fetchData = async () => {
      const stats = await InteractionService.getWeeklyStats();
      setGardenState({
        positiveCount: stats.positive,
        negativeCount: stats.negative,
        ratio: stats.ratio,
        healthLevel: calculateHealthLevel(stats.ratio)
      });
    };
    fetchData();
  }, []);

  const calculateHealthLevel = (ratio) => {
    const targetRatio = 5;
    return Math.min(100, (ratio / targetRatio) * 100);
  };

  const handleAddInteraction = async (type) => {
    await InteractionService.addInteraction(type);
    // Update local state
    const newPositive = type === 'flower' ? gardenState.positiveCount + 1 : gardenState.positiveCount;
    const newNegative = type === 'weed' ? gardenState.negativeCount + 1 : gardenState.negativeCount;
    const newRatio = newPositive / Math.max(1, newNegative);
    
    setGardenState({
      positiveCount: newPositive,
      negativeCount: newNegative,
      ratio: newRatio,
      healthLevel: calculateHealthLevel(newRatio)
    });
  };

  return (
    <div className="min-h-screen bg-gradient-to-b from-blue-50 to-green-50">
      <div className="container mx-auto px-4 py-8">
        <h1 className="text-3xl font-bold text-center text-green-800 mb-6">
          باغ عاطفی شما
        </h1>
        
        <div className="grid grid-cols-1 md:grid-cols-3 gap-8">
          {/* بخش گرافیکی باغ */}
          <div className="md:col-span-2">
            <GardenVisualization healthLevel={gardenState.healthLevel} />
          </div>
          
          {/* بخش تعاملات */}
          <div className="space-y-6">
            <div className="bg-white p-6 rounded-lg shadow-md">
              <h2 className="text-xl font-semibold mb-4">نسبت عاطفی شما</h2>
              <div className="h-4 w-full bg-gray-200 rounded-full overflow-hidden">
                <div 
                  className="h-full bg-green-500 transition-all duration-500"
                  style={{ width: `${gardenState.healthLevel}%` }}
                ></div>
              </div>
              <p className="mt-2 text-gray-700">
                نسبت فعلی: {gardenState.ratio.toFixed(1)} به ۱
              </p>
            </div>

            <div className="flex space-x-4">
              <button
                onClick={() => handleAddInteraction('flower')}
                className="flex-1 bg-green-500 hover:bg-green-600 text-white py-3 px-4 rounded-lg shadow-md transition-colors"
              >
                کاشت گل 🌸
              </button>
              <button
                onClick={() => handleAddInteraction('weed')}
                className="flex-1 bg-red-500 hover:bg-red-600 text-white py-3 px-4 rounded-lg shadow-md transition-colors"
              >
                علف هرز 🌿
              </button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}

export default Garden;
```

### 3. کامپوننت گرافیکی باغ (GardenVisualization.js)
```javascript
import React from 'react';

const GardenVisualization = ({ healthLevel }) => {
  const getGardenState = () => {
    if (healthLevel > 80) return 'flourishing';
    if (healthLevel > 50) return 'healthy';
    if (healthLevel > 30) return 'struggling';
    return 'wilting';
  };

  const gardenState = getGardenState();

  return (
    <div className="relative h-96 bg-green-100 rounded-xl overflow-hidden border-4 border-green-300">
      {/* زمین */}
      <div className="absolute bottom-0 w-full h-1/3 bg-green-700"></div>
      
      {/* عناصر باغ بر اساس سلامت */}
      {gardenState === 'flourishing' && (
        <>
          <div className="absolute bottom-1/3 left-1/4 w-16 h-16 bg-green-500 rounded-full"></div>
          <div className="absolute bottom-1/3 right-1/4 w-16 h-16 bg-pink-300 rounded-full"></div>
          <div className="absolute bottom-1/2 left-1/2 transform -translate-x-1/2 w-24 h-24 bg-green-600 rounded-full"></div>
        </>
      )}
      
      {/* انیمیشن‌های ساده با CSS */}
      <style jsx>{`
        @keyframes sway {
          0%, 100% { transform: rotate(-2deg); }
          50% { transform: rotate(2deg); }
        }
        .flower {
          animation: sway 3s ease-in-out infinite;
        }
      `}</style>
    </div>
  );
};

export default GardenVisualization;
```

## بخش بک‌اند (Node.js)

### 1. مدل تعاملات (interaction.model.js)
```javascript
const mongoose = require('mongoose');

const InteractionSchema = new mongoose.Schema({
  coupleId: {
    type: mongoose.Schema.Types.ObjectId,
    ref: 'Couple',
    required: true
  },
  type: {
    type: String,
    enum: ['flower', 'weed'],
    required: true
  },
  description: {
    type: String,
    maxlength: 200
  },
  partner: {
    type: String,
    enum: ['partner1', 'partner2'],
    required: true
  },
  timestamp: {
    type: Date,
    default: Date.now
  }
});

module.exports = mongoose.model('Interaction', InteractionSchema);
```

### 2. کنترلر تعاملات (interaction.controller.js)
```javascript
const Interaction = require('../models/interaction.model');

exports.addInteraction = async (req, res) => {
  try {
    const { type, description, partner } = req.body;
    const newInteraction = new Interaction({
      coupleId: req.user.coupleId,
      type,
      description,
      partner
    });
    await newInteraction.save();
    res.status(201).json({ message: 'Interaction recorded successfully' });
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

exports.getStats = async (req, res) => {
  try {
    const coupleId = req.user.coupleId;
    const [positive, negative] = await Promise.all([
      Interaction.countDocuments({ coupleId, type: 'flower' }),
      Interaction.countDocuments({ coupleId, type: 'weed' })
    ]);
    const ratio = positive / Math.max(1, negative);
    res.json({ positive, negative, ratio });
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};
```

## راه‌اندازی و استقرار

1. **پیش‌نیازها**:
   - Node.js v16+
   - MongoDB Atlas یا محلی
   - Git

2. **نصب**:
```bash
# Clone the repository
git clone https://github.com/yourusername/gottman-therapy-app.git
cd gottman-therapy-app

# Install dependencies
cd client && npm install
cd ../server && npm install
```

3. **پیکربندی**:
   - ایجاد فایل `.env` در پوشه server با محتوای:
     ```
     MONGODB_URI=your_mongodb_connection_string
     JWT_SECRET=your_jwt_secret_key
     ```

4. **اجرا**:
```bash
# Run backend
cd server && npm start

# Run frontend (in another terminal)
cd client && npm start
```

5. **استقرار**:
   - فرانت‌اند: Vercel یا Netlify
   - بک‌اند: Heroku یا Railway
   - دیتابیس: MongoDB Atlas

## نکات طراحی کلیدی:

1. **تئوری گاتمن**:
   - نسبت ۵ تعامل مثبت به ۱ تعامل منفی برای سلامت رابطه
   - تعاملات مثبت شامل: قدردانی، محبت، شنونده خوب بودن
   - تعاملات منفی شامل: انتقاد، تحقیر، دفاعی بودن، سنگ‌اندازی

2. **گیمیفیکیشن**:
   - پیشرفت بصری (باغ شکوفا در مقابل پژمرده)
   - چالش‌های روزانه/هفتگی
   - پاداش‌های غیرمادی (انیمیشن، نشان‌ها)

3. **امنیت**:
   - رمزنگاری داده‌های حساس
   - احراز هویت دو مرحله‌ای
   - دسترسی درمانگر فقط با اجازه کاربران

4. **تجربه کاربری**:
   - رابط ساده و غیررسمی
   - بازخورد فوری برای تعاملات
   - گزارش‌های قابل فهم
